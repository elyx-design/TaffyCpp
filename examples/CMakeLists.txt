cmake_minimum_required(VERSION 3.25)
project(taffy_examples LANGUAGES CXX)
find_package(taffy_cpp CONFIG REQUIRED)
enable_testing()
foreach(example flex grid measurement mutation custom_tree)
    add_executable(${example} ${example}.cpp)
    target_link_libraries(${example} PRIVATE taffy::taffy)
    target_compile_options(${example} PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic;-Werror>)
    add_test(NAME ${example} COMMAND ${example})
endforeach()
